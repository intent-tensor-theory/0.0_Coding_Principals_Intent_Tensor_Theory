# 0.1.c.i Template: State Engine Architecture

**Purpose:** Reference template for creating Ghostless state engines across any domain.

---

## State Engine Pattern

```typescript
// [Level].[sublevel]_file[Domain]StateEngine.ts

type [Domain]StateType = {
  // All state properties for this domain
  [property]: [type];
};

export const intentStateEngine[Domain][Level] = {
  // State properties
  [stateProperty]: [defaultValue] as [Type],

  // Mutation functions (explicit, logged)
  set[PropertyName]([param]: [Type]): void {
    this.[stateProperty] = [param];
    logger.log('📊 [FILE_PATH].set[PropertyName]: ' +
      'Set [stateProperty]=' + [param]);
  },

  // Computed getters (if needed)
  get[ComputedProperty](): [Type] {
    return [computation];
  },

  // Reset function
  reset[Domain]State(): void {
    Object.keys(this).forEach(key => {
      if (typeof this[key] !== 'function') {
        this[key] = [defaultValue];
      }
    });
    logger.log('📊 [FILE_PATH].reset[Domain]State: State reset');
  }
};
```

## Naming Conventions

- **Engine Name:** `intentStateEngine[Domain][Level]`
- **File Name:** `[level].[sublevel]_file[Domain]StateEngine.ts`
- **Mutators:** `set[PropertyName]()`
- **Getters:** `get[ComputedProperty]()`
- **State Properties:** `[domain][Descriptor][Context]`

## Example: Dashboard State Engine

```typescript
// 1.0.a_fileDashboardStateEngine.ts

export const intentStateEngineDashboardPrimary = {
  selectedViewMode: 'grid' as 'grid' | 'list',
  activeFilters: [] as string[],
  searchQuery: '' as string,
  isLoadingDashboard: false as boolean,

  setSelectedViewMode(mode: 'grid' | 'list'): void {
    this.selectedViewMode = mode;
    logger.log('📊 1.0.a_fileDashboardStateEngine.setSelectedViewMode: mode=' + mode);
  },

  setActiveFilters(filters: string[]): void {
    this.activeFilters = filters;
    logger.log('📊 1.0.a_fileDashboardStateEngine.setActiveFilters: count=' + filters.length);
  }
};
```
