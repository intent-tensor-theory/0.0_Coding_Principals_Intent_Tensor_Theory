## Validation Pattern

```typescript
// [Level].[sublevel]_fileValidate[Object][Domain]Intent.ts

export function validate[Object][Domain]By[Context][Level](
  [dataToValidate]: [InputType]
): [OutputType]Validated | ValidationError {

  logger.log('✅ [FILE_PATH].validate[Object][Domain]By[Context][Level]: ' +
    'Validating [object] with [context]');

  const validationErrors: string[] = [];

  // Validation checks
  if ([condition]) {
    validationErrors.push('[error message]');
    logger.log('⚠️ [FILE_PATH].validate[Object][Domain]: Validation failed - [reason]');
  }

  if (validationErrors.length > 0) {
    logger.log('❌ [FILE_PATH].validate[Object][Domain]: Validation failed with ' +
      validationErrors.length + ' errors');

    return {
      isValid: false,
      errors: validationErrors
    };
  }

  logger.log('✅ [FILE_PATH].validate[Object][Domain]: Validation passed');

  return {
    isValid: true,
    validatedData: [dataToValidate] as [OutputType]Validated
  };
}
```

## Naming Conventions

- **Function Name:** `validate[Object][Domain]By[Context][Level]()`
- **File Name:** `[level].[sublevel]_fileValidate[Object][Domain]Intent.ts`
- **Input Type:** `[Object]DataType`
- **Output Type:** `[Object]DataTypeValidated`
- **Return:** `ValidationResult` with isValid + errors/validatedData

## Example: Email Validation

```typescript
// 0.0.b_fileValidateEmailAddressIntent.ts

type EmailValidationResult = {
  isValid: boolean;
  errors?: string[];
  validatedEmail?: string;
};

export function validateEmailAddressStringByFormat(
  emailString: string
): EmailValidationResult {

  logger.log('✅ 0.0.b_fileValidateEmailAddress.validateEmailAddressStringByFormat: ' +
    'Validating email=' + emailString);

  const validationErrors: string[] = [];

  // Check if empty
  if (!emailString || emailString.trim() === '') {
    validationErrors.push('Email cannot be empty');
  }

  // Check format
  const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
  if (!emailRegex.test(emailString)) {
    validationErrors.push('Email format is invalid');
    logger.log('⚠️ 0.0.b_fileValidateEmailAddress: Invalid email format');
  }

  // Check length
  if (emailString.length > 254) {
    validationErrors.push('Email exceeds maximum length (254 characters)');
  }

  if (validationErrors.length > 0) {
    logger.log('❌ 0.0.b_fileValidateEmailAddress: Validation failed with ' +
      validationErrors.length + ' errors');

    return {
      isValid: false,
      errors: validationErrors
    };
  }

  logger.log('✅ 0.0.b_fileValidateEmailAddress: Email validated successfully');

  return {
    isValid: true,
    validatedEmail: emailString.trim().toLowerCase()
  };
}
```

## Example: Form Data Validation

```typescript
// 2.3.c_fileValidatePlatformFormData.ts

type PlatformFormDataType = {
  platformName: string;
  platformDomain: string;
  contactEmail: string;
};

type PlatformFormDataTypeValidated = PlatformFormDataType & {
  validatedAt: Date;
  validatedBy: string;
};

export function validatePlatformFormDataByRequiredFieldsLevelTwo(
  formData: PlatformFormDataType
): { isValid: boolean; errors?: string[]; validatedData?: PlatformFormDataTypeValidated } {

  logger.log('✅ 2.3.c_fileValidatePlatformFormData.validatePlatformFormDataByRequiredFieldsLevelTwo: ' +
    'Validating platform form data');

  const validationErrors: string[] = [];

  // Validate platform name
  if (!formData.platformName || formData.platformName.trim().length < 3) {
    validationErrors.push('Platform name must be at least 3 characters');
  }

  // Validate domain
  const domainRegex = /^[a-zA-Z0-9][a-zA-Z0-9-]{1,61}[a-zA-Z0-9]\.[a-zA-Z]{2,}$/;
  if (!domainRegex.test(formData.platformDomain)) {
    validationErrors.push('Invalid domain format');
    logger.log('⚠️ 2.3.c_fileValidatePlatformFormData: Invalid domain format');
  }

  // Validate email
  const emailValidation = validateEmailAddressStringByFormat(formData.contactEmail);
  if (!emailValidation.isValid) {
    validationErrors.push('Contact email is invalid');
  }

  if (validationErrors.length > 0) {
    logger.log('❌ 2.3.c_fileValidatePlatformFormData: Validation failed with ' +
      validationErrors.length + ' errors');

    return {
      isValid: false,
      errors: validationErrors
    };
  }

  logger.log('✅ 2.3.c_fileValidatePlatformFormData: Validation passed');

  return {
    isValid: true,
    validatedData: {
      ...formData,
      validatedAt: new Date(),
      validatedBy: 'system'
    }
  };
}
```
