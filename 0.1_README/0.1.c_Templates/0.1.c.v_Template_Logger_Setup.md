## Logger Pattern

```typescript
// 0.2.3_logging/0.2.3.a_fileLoggerCore.ts

type LogLevel = 'info' | 'warning' | 'error' | 'debug';

type LogEntry = {
  timestamp: Date;
  level: LogLevel;
  file: string;
  function: string;
  message: string;
  emoji: string;
};

export const logger = {
  logHistory: [] as LogEntry[],

  log(message: string, level: LogLevel = 'info'): void {
    const emoji = this.getEmojiForMessage(message);
    const timestamp = new Date();

    const logEntry: LogEntry = {
      timestamp,
      level,
      file: this.extractFileFromMessage(message),
      function: this.extractFunctionFromMessage(message),
      message,
      emoji
    };

    this.logHistory.push(logEntry);

    // Output to console with emoji
    console.log(`[${timestamp.toISOString()}] ${emoji} ${message}`);

    // Optional: Send to external logging service
    if (level === 'error') {
      this.sendToErrorTracking(logEntry);
    }
  },

  getEmojiForMessage(message: string): string {
    // Map emoji based on message content
    if (message.includes('🔐') || message.includes('Auth')) return '🔐';
    if (message.includes('📊') || message.includes('State')) return '📊';
    if (message.includes('🎨') || message.includes('Render')) return '🎨';
    if (message.includes('🎯') || message.includes('handle')) return '🎯';
    if (message.includes('✅') || message.includes('Validat')) return '✅';
    if (message.includes('🚨') || message.includes('ERROR')) return '🚨';
    if (message.includes('⚠️') || message.includes('Warning')) return '⚠️';
    if (message.includes('💰') || message.includes('Payment')) return '💰';
    if (message.includes('📧') || message.includes('Email')) return '📧';
    return '📝';
  },

  extractFileFromMessage(message: string): string {
    const match = message.match(/(\d+\.\d+\.[\w]+_file[\w]+)/);
    return match ? match[1] : 'unknown';
  },

  extractFunctionFromMessage(message: string): string {
    const match = message.match(/\.(\w+):/);
    return match ? match[1] : 'unknown';
  },

  sendToErrorTracking(logEntry: LogEntry): void {
    // Integration with Sentry, LogRocket, etc.
    // Example: Sentry.captureMessage(logEntry.message);
  },

  getLogsByFile(fileName: string): LogEntry[] {
    return this.logHistory.filter(entry => entry.file === fileName);
  },

  getLogsByLevel(level: LogLevel): LogEntry[] {
    return this.logHistory.filter(entry => entry.level === level);
  },

  exportLogs(): string {
    return JSON.stringify(this.logHistory, null, 2);
  }
};
```

## Log Format Standard

**Format:** `[emoji] [file].[function]: [description] [key=value pairs]`

**Examples:**
```
🔐 0.0.a_fileAuthValidationCore.validateAuthTokenByRequest: Token validated for userId=abc123
📊 1.0.a_fileDashboardStateEngine.setSelectedViewMode: Set selectedViewMode=grid
🎨 3.3.a_fileRenderContributorsIntent.renderListContributorsIntentByPlatformIdLevelFour: Rendering 5 contributors for platformId=xyz789
🚨 2.5.a_filePaymentProcessIntent.processPaymentByPaymentId: Payment failed with error=insufficient_funds
```
