# 0.1.d.iv ISA Validator Specification

**Purpose:** Automated validator for Infrastructure Semantic Anchoring (ISA) compliance.

---

## What Does ISA Validator Check?

Validates that all infrastructure resources follow ISA naming conventions:
- Platform anchors (`gcp-`, `aws-`, `git-`, etc.)
- Semantic chains (project → architecture → component → environment)
- Cross-platform reference consistency

## Validator Requirements

### Input
- Infrastructure manifest (JSON/YAML)
- Cloud platform APIs (GCP, AWS, etc.)
- Configuration files (.env, config.json)

### Output
```json
{
  "schemaVersion": "1.0.0",
  "validatedAt": "2025-01-27T10:00:00Z",
  "totalResources": 50,
  "compliantResources": 42,
  "violations": 8,
  "complianceRate": "84%",
  "status": "NEEDS_IMPROVEMENT",
  "violations": [ /* specific violations */ ],
  "recommendations": [ /* fix suggestions */ ]
}
```

## Implementation Spec

```typescript
// isaValidator.ts

interface ISAValidationConfig {
  infrastructureManifest?: string;
  scanGCP?: boolean;
  scanAWS?: boolean;
  scanGitHub?: boolean;
  configFiles?: string[];
}

interface ISAViolation {
  resourceType: 'project' | 'oauth' | 'service-account' | 'storage' | 'repo';
  platform: string;
  currentName: string;
  issue: string;
  suggestedName: string;
  severity: 'high' | 'medium' | 'low';
}

interface ISAValidationReport {
  schemaVersion: string;
  validatedAt: string;
  totalResources: number;
  compliantResources: number;
  violations: number;
  complianceRate: string;
  status: 'EXCELLENT' | 'GOOD' | 'NEEDS_IMPROVEMENT' | 'CRITICAL';
  violationsList: ISAViolation[];
  recommendations: string[];
}

class ISAValidator {
  private config: ISAValidationConfig;
  private violations: ISAViolation[] = [];
  private totalResources = 0;

  async validate(): Promise<ISAValidationReport> {
    console.log('🔗 ISAValidator: Starting validation...');

    // Scan infrastructure
    await this.scanInfrastructure();

    // Validate naming
    await this.validateNaming();

    // Check cross-references
    await this.validateCrossReferences();

    // Generate report
    const report = this.generateReport();

    console.log(`🔗 ISAValidator: Found ${report.violations} violations (${report.complianceRate} compliant)`);

    return report;
  }

  private async scanInfrastructure(): Promise<void> {
    if (this.config.scanGCP) {
      await this.scanGCPResources();
    }

    if (this.config.scanAWS) {
      await this.scanAWSResources();
    }

    if (this.config.scanGitHub) {
      await this.scanGitHubResources();
    }
  }

  private async scanGCPResources(): Promise<void> {
    // Use GCP API to list:
    // - Projects
    // - OAuth clients
    // - Service accounts
    // - Storage buckets
    // - Cloud Functions
  }

  private async validateNaming(): Promise<void> {
    // Check each resource name
    // Validate format: [platform]-[project]-[architecture]-[component]-[env]
  }

  private isISACompliant(resourceName: string, resourceType: string): boolean {
    // Check platform anchor
    const validAnchors = ['gcp-', 'aws-', 'azure-', 'git-', 'vercel-', 'supabase-', 'firebase-'];
    const hasAnchor = validAnchors.some(anchor => resourceName.startsWith(anchor));

    if (!hasAnchor) {
      return false;
    }

    // Check semantic completeness
    const parts = resourceName.split('-');
    if (parts.length < 2) {
      return false; // Must have at least [platform]-[project]
    }

    return true;
  }

  private generateSuggestedName(
    currentName: string,
    resourceType: string,
    platform: string
  ): string {
    // Suggest ISA-compliant name
    // Example: "my-app" → "gcp-my-app-api-service"

    // Extract potential project name
    const projectName = currentName.replace(/[^a-z0-9-]/g, '-');

    // Suggest based on resource type
    const suggestions = {
      'project': `${platform}-${projectName}`,
      'oauth': `${platform}-${projectName}-oauth-prod`,
      'service-account': `${platform}-${projectName}-service-account-prod`,
      'storage': `${platform}-${projectName}-storage-bucket`,
      'repo': `git-${projectName}`
    };

    return suggestions[resourceType] || `${platform}-${projectName}`;
  }

  private async validateCrossReferences(): Promise<void> {
    // Check that config files reference correct ISA names
    // Example: Verify .env has correct resource names
  }

  private generateReport(): ISAValidationReport {
    const compliantResources = this.totalResources - this.violations.length;
    const complianceRate = ((compliantResources / this.totalResources) * 100).toFixed(0);

    const status =
      parseInt(complianceRate) >= 95 ? 'EXCELLENT' :
      parseInt(complianceRate) >= 80 ? 'GOOD' :
      parseInt(complianceRate) >= 60 ? 'NEEDS_IMPROVEMENT' :
      'CRITICAL';

    return {
      schemaVersion: '1.0.0',
      validatedAt: new Date().toISOString(),
      totalResources: this.totalResources,
      compliantResources,
      violations: this.violations.length,
      complianceRate: complianceRate + '%',
      status,
      violationsList: this.violations,
      recommendations: this.generateRecommendations()
    };
  }

  private generateRecommendations(): string[] {
    const recs: string[] = [];

    if (this.violations.length > 0) {
      recs.push('Rename resources to follow ISA naming conventions');
      recs.push('Update all configuration files to reference new names');
      recs.push('Use ISA naming for all new infrastructure');
    }

    const highSeverity = this.violations.filter(v => v.severity === 'high');
    if (highSeverity.length > 0) {
      recs.push(`Focus on ${highSeverity.length} high-severity violations first (OAuth, service accounts)`);
    }

    return recs;
  }
}

// Usage
const validator = new ISAValidator({
  scanGCP: true,
  scanGitHub: true,
  configFiles: ['.env', 'config.json']
});

const report = await validator.validate();
console.log(report);
```

## CLI Usage

```bash
# Validate all infrastructure
isa-validator validate --gcp --aws --github

# Validate with auto-fix suggestions
isa-validator validate --gcp --suggest-renames

# Validate specific resources
isa-validator validate --resource gcp-project-12345

# Export report
isa-validator validate --gcp --output isa-report.json

# Check config files
isa-validator check-config .env config.json
```
