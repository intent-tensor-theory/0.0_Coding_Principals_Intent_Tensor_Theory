## 📖 THE DECISION-AT-RUNTIME PROBLEM

Traditional code makes decisions during execution:

```javascript
// Traditional runtime decision-making
function processRows(rows) {
  let processed = 0;
  let skipped = 0;

  for (const row of rows) {
    // Decision happens DURING execution
    if (row.domain && row.domain !== 'Attempted' && row.result === '') {
      try {
        const result = fetchData(row.domain);
        writeResult(row, result);
        processed++;
      } catch (error) {
        skipped++;
      }
    } else {
      skipped++;
    }
  }

  console.log(`Processed: ${processed}, Skipped: ${skipped}`);
}
```

**Problems:**
- Decisions mixed with actions
- No preview of what will happen
- Errors discovered mid-execution
- No audit trail before execution
- Impossible to validate logic before running

**MetaMap Schema solves this through complete precomputation.**

---

## 1.0 WHAT IS A METAMAP?

**MetaMap:** A structured data object that makes ALL decisions about data eligibility and processing BEFORE any action is taken.

### Core Principle

```
PRE-X = Precomputed Execution
```

The MetaMap is the **contract**. The execution kernel **follows** the contract.

**The kernel never decides — it obeys.**

## 2.0 METAMAP STRUCTURE

### Basic Schema

```typescript
interface MetaMap<T> {
  schemaVersion: string;
  generatedAt: string;
  totalRows: number;
  processableRows: number;
  skippedRows: number;
  rows: Record<string, MetaMapRow<T>>;
}

interface MetaMapRow<T> {
  rowIndex: number;
  data: T;
  metaTags: MetaTags;
}

interface MetaTags {
  [key: string]: boolean;
  isRowProcessable: boolean; // THE LYNCHPIN TAG
}
```

## 3.0 METATAGS: THE DECISION LAYER

### The Lynchpin Pattern

Every MetaMap has ONE final boolean that determines processability:

```typescript
metaTags: {
  // Individual conditions
  isEmailPopulated: true,
  isEmailValid: true,
  isDomainResolvable: true,
  isStatusWritable: true,

  // LYNCHPIN: The converged decision
  isRowProcessable: isEmailPopulated &&
                    isEmailValid &&
                    isDomainResolvable &&
                    isStatusWritable
}
```

**The lynchpin tag is the ONLY tag the kernel reads.**

## 4.0 METAMAP GENERATION PATTERN

### The Generator Function

```typescript
function generateMetaMap<T>(
  rawData: T[],
  config: MetaMapConfig
): MetaMap<T> {
  logger.log('🗺️ generateMetaMap: Starting generation for ' + rawData.length + ' rows');

  const metaMap: MetaMap<T> = {
    schemaVersion: '1.0.0',
    generatedAt: new Date().toISOString(),
    totalRows: rawData.length,
    processableRows: 0,
    skippedRows: 0,
    rows: {}
  };

  rawData.forEach((data, index) => {
    // Generate metaTags for this row
    const metaTags = generateMetaTagsForRow(data, config);

    // Store in MetaMap
    metaMap.rows[index] = {
      rowIndex: index,
      data,
      metaTags
    };

    // Count processable rows
    if (metaTags.isRowProcessable) {
      metaMap.processableRows++;
    } else {
      metaMap.skippedRows++;
    }
  });

  logger.log('✅ generateMetaMap: Generated MetaMap with ' +
    metaMap.processableRows + ' processable rows, ' +
    metaMap.skippedRows + ' skipped');

  return metaMap;
}
```

### The MetaTags Generator

```typescript
function generateMetaTagsForRow<T>(
  data: T,
  config: MetaMapConfig
): MetaTags {
  // Example: Email processing
  const metaTags: any = {};

  // Check 1: Email populated
  metaTags.isEmailPopulated = data.email && data.email.trim() !== '';

  // Check 2: Email valid
  const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
  metaTags.isEmailValid = emailRegex.test(data.email);

  // Check 3: Status not attempted
  metaTags.isStatusNotAttempted = data.status !== 'Attempted';

  // Check 4: Result writable
  metaTags.isResultWritable = data.result === '';

  // LYNCHPIN: Converge all checks
  metaTags.isRowProcessable =
    metaTags.isEmailPopulated &&
    metaTags.isEmailValid &&
    metaTags.isStatusNotAttempted &&
    metaTags.isResultWritable;

  return metaTags;
}
```

## 5.0 PRE-X EXECUTION KERNEL

### The Kernel Reads the MetaMap

```typescript
function executeMetaMapPREX<T>(
  metaMap: MetaMap<T>,
  processFunction: (data: T) => Promise<any>
): Promise<void> {
  logger.log('⚙️ executeMetaMapPREX: Starting execution for ' +
    metaMap.processableRows + ' processable rows');

  // Filter to only processable rows
  const processableRows = Object.values(metaMap.rows)
    .filter(row => row.metaTags.isRowProcessable);

  logger.log('⚙️ executeMetaMapPREX: Filtered to ' + processableRows.length + ' rows');

  // Execute in batches
  return processBatch(processableRows, processFunction);
}

async function processBatch<T>(
  rows: MetaMapRow<T>[],
  processFunction: (data: T) => Promise<any>
): Promise<void> {
  for (const row of rows) {
    try {
      logger.log('⚙️ processBatch: Processing row ' + row.rowIndex);
      await processFunction(row.data);
    } catch (error) {
      logger.log('🚨 processBatch: Error processing row ' + row.rowIndex + ': ' + error.message);
    }
  }
}
```

### Key Insight: The Kernel Never Decides

```typescript
// ❌ WRONG: Kernel makes decisions
function badExecute(rows) {
  for (const row of rows) {
    if (row.email && row.email.includes('@') && row.status !== 'Attempted') {
      // Kernel is deciding!
      processRow(row);
    }
  }
}

// ✅ CORRECT: Kernel follows MetaMap
function goodExecute(metaMap) {
  const processableRows = Object.values(metaMap.rows)
    .filter(row => row.metaTags.isRowProcessable); // MetaMap already decided

  processableRows.forEach(row => processRow(row.data));
}
```

```typescript
interface CampaignRow {

function generateEmailCampaignMetaMap(
  rows: CampaignRow[]
): MetaMap<CampaignRow> {
  logger.log('📧 generateEmailCampaignMetaMap: Generating for ' + rows.length + ' contacts');

  const metaMap: MetaMap<CampaignRow> = {
    schemaVersion: '1.0.0',
    generatedAt: new Date().toISOString(),
    totalRows: rows.length,
    processableRows: 0,
    skippedRows: 0,
    rows: {}

  rows.forEach((row, index) => {
    const metaTags = {
      isEmailPopulated: row.email && row.email.trim() !== '',
      isEmailValid: /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(row.email),
      isOptedIn: row.optIn === true,
      isNotPreviouslySent: row.previouslySent === false,

      // LYNCHPIN
      isRowProcessable:
        row.email && row.email.trim() !== '' &&
        /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(row.email) &&
        row.optIn === true &&
        row.previouslySent === false
    };

    metaMap.rows[index] = {
      rowIndex: index,
      data: row,
      metaTags
    };

    if (metaTags.isRowProcessable) {
      metaMap.processableRows++;
    } else {
      metaMap.skippedRows++;
    }
  });

  logger.log('✅ MetaMap: ' + metaMap.processableRows + ' will be sent, ' +
    metaMap.skippedRows + ' will be skipped');

  return metaMap;
}
```

```typescript
function reviewMetaMapBeforeSending(metaMap: MetaMap<CampaignRow>): void {
```

## 7.0 METAMAP FALLBACK PATTERNS

### Pattern: Primary + Fallback Logic

```typescript
interface RowWithFallback {
  domain: string;
  fallbackEmails: string[];
  result: string;
}

function generateMetaMapWithFallback(
  rows: RowWithFallback[]
): MetaMap<RowWithFallback> {
  const metaMap: MetaMap<RowWithFallback> = {
    schemaVersion: '1.0.0',
    generatedAt: new Date().toISOString(),
    totalRows: rows.length,
    processableRows: 0,
    skippedRows: 0,
    rows: {}
  };

  rows.forEach((row, index) => {
    const metaTags = {
      // Primary checks
      isDomainPopulated: row.domain && row.domain.trim() !== '',
      isDomainFetchable: isValidDomain(row.domain),
      isResultWritable: row.result === '',

      // Fallback checks
      hasFallbackEmails: row.fallbackEmails && row.fallbackEmails.length > 0,
      areFallbackEmailsValid: row.fallbackEmails?.every(email =>
        /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)
      ),

      // Composite: Can process with primary OR fallback
      canProcessWithPrimary:
        row.domain && row.domain.trim() !== '' &&
        isValidDomain(row.domain) &&
        row.result === '',

      canProcessWithFallback:
        !(row.domain && isValidDomain(row.domain)) &&
        row.fallbackEmails && row.fallbackEmails.length > 0 &&
        row.fallbackEmails.every(email => /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) &&
        row.result === '',

      // LYNCHPIN: Either primary OR fallback works
      isRowProcessable:
        (row.domain && isValidDomain(row.domain) || (row.fallbackEmails?.length > 0)) &&
        row.result === ''
    };

    metaMap.rows[index] = { rowIndex: index, data: row, metaTags };

    if (metaTags.isRowProcessable) {
      metaMap.processableRows++;
    } else {
      metaMap.skippedRows++;
    }
  });

  return metaMap;
}
```

## 9.0 METAMAP VALIDATION & TESTING

### Validating MetaTags Logic

```typescript
function validateMetaMapLogic(metaMap: MetaMap<any>): boolean {
  let valid = true;

  Object.values(metaMap.rows).forEach(row => {
    // Verify lynchpin tag is properly computed
    const recomputedProcessable = Object.entries(row.metaTags)
      .filter(([key]) => key !== 'isRowProcessable')
      .every(([_, value]) => value === true);

    if (row.metaTags.isRowProcessable !== recomputedProcessable) {
      console.error('❌ Row ' + row.rowIndex + ': isRowProcessable mismatch');
      valid = false;
    }
  });

  return valid;
}
```

### Caching MetaMaps

```typescript
const metaMapCache = new Map<string, MetaMap<any>>();

function getCachedMetaMap<T>(
  cacheKey: string,
  generator: () => MetaMap<T>
): MetaMap<T> {
  if (metaMapCache.has(cacheKey)) {
    logger.log('🔄 getCachedMetaMap: Using cached MetaMap for ' + cacheKey);
    return metaMapCache.get(cacheKey)!;
  }

  logger.log('🗺️ getCachedMetaMap: Generating new MetaMap for ' + cacheKey);
  const metaMap = generator();
  metaMapCache.set(cacheKey, metaMap);

  return metaMap;
}
```

## 11.0 METAMAP ERROR HANDLING

### Handling MetaMap Generation Errors

```typescript
function safeGenerateMetaMap<T>(
  rawData: T[],
  config: MetaMapConfig
): MetaMap<T> | { error: string } {
  try {
    return generateMetaMap(rawData, config);
  } catch (error) {
    logger.log('🚨 safeGenerateMetaMap: Generation failed: ' + error.message);

    return {
      error: 'MetaMap generation failed: ' + error.message
    };
  }
}
```

### Handling Execution Errors

```typescript
async function safeExecuteMetaMap<T>(
  metaMap: MetaMap<T>,
  processFunction: (data: T) => Promise<any>
): Promise<{ successful: number; failed: number }> {
  const results = { successful: 0, failed: 0 };

  const processableRows = Object.values(metaMap.rows)
    .filter(row => row.metaTags.isRowProcessable);

  for (const row of processableRows) {
    try {
      await processFunction(row.data);
      results.successful++;
    } catch (error) {
      logger.log('🚨 Row ' + row.rowIndex + ' failed: ' + error.message);
      results.failed++;
    }
  }

  logger.log('✅ Execution complete: ' +
    results.successful + ' successful, ' +
    results.failed + ' failed');

  return results;
}
```

### 12.2 Incremental MetaMaps

```typescript
function updateMetaMapIncremental<T>(
  existingMetaMap: MetaMap<T>,
  newRows: T[],
  config: MetaMapConfig
): MetaMap<T> {
  logger.log('🔄 updateMetaMapIncremental: Adding ' + newRows.length + ' new rows');

  const updatedMetaMap = { ...existingMetaMap };

  newRows.forEach((data, index) => {
    const rowIndex = existingMetaMap.totalRows + index;
    const metaTags = generateMetaTagsForRow(data, config);

    updatedMetaMap.rows[rowIndex] = { rowIndex, data, metaTags };

    if (metaTags.isRowProcessable) {
      updatedMetaMap.processableRows++;
    } else {
      updatedMetaMap.skippedRows++;
    }
  });

  updatedMetaMap.totalRows += newRows.length;

  return updatedMetaMap;
}
```
